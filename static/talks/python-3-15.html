01Lazy imports

# top of your module
__lazy_modules__ = ("numpy", "pandas", "torch")

import numpy   # <- not actually imported yet
import pandas

def crunch():
    # numpy is materialised here, on first use
    return numpy.zeros(1000)
>>> import sys, my_module
>>> sys.lazy_modules
{'numpy': <lazy import>,
 'pandas': <lazy import>,
 'torch': <lazy import>}

>>> my_module.crunch()  # numpy lazily imported here
array([0., 0., 0., ..., 0.])
